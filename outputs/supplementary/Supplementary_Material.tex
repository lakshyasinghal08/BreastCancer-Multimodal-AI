\section{Mathematical Invariance Proof}
Let $\mathcal{A} \subseteq \{1, \dots, 7\}$ index observed modalities. For any query token $i$, the dynamic mask-aware attention weight for token $j$ is:
\begin{equation}
\alpha_{i, j} = \frac{\exp\left( \frac{q_i k_j^T}{\sqrt{d_k}} + (S_{\text{mask}})_{i, j} \right)}{\sum_{m=0}^7 \exp\left( \frac{q_i k_m^T}{\sqrt{d_k}} + (S_{\text{mask}})_{i, m} \right)}
\end{equation}
When $(S_{\text{mask}})_{i, j} = -\infty$ for $j \notin \mathcal{A}$, $\exp(-\infty) = 0$, strictly eliminating unobserved modalities from the attention distribution without synthetic imputation bias.

\section{Calibration and Uncertainty Quantification}
Post-hoc temperature scaling ($T^* = 1.18$) calibrates output logits:
\begin{equation}
\hat{p} = \sigma\left(\frac{z}{1.18}\right)
\end{equation}
Predictive uncertainty is quantified via Shannon entropy:
\begin{equation}
H(\hat{p}) = -\left[\hat{p}\log_2(\hat{p}) + (1-\hat{p})\log_2(1-\hat{p})\right]
\end{equation}
which correlates strongly with misclassification ($r = 0.682, p < 10^{-15}$).
